% Compact two-column header shared by Chinese and English documents.
\usepackage{graphicx}
% Upload photo.jpg or photo.png to the same directory as main.tex.
% Photos are scaled proportionally to fit within 2.2 cm x 2.9 cm.
\newcommand{\cvphotobox}{%
  \IfFileExists{photo.jpg}{%
    \includegraphics[width=2.2cm,height=2.9cm,keepaspectratio]{photo.jpg}%
  }{%
    \IfFileExists{photo.png}{%
      \includegraphics[width=2.2cm,height=2.9cm,keepaspectratio]{photo.png}%
    }{%
      {\setlength{\fboxsep}{0pt}\setlength{\fboxrule}{0.3pt}\color{accent!25}%
      \fbox{\parbox[c][2.9cm][c]{2.2cm}{\centering
        \normalfont\scriptsize\color{accent!65}PHOTO\\[3pt]
        photo.jpg / photo.png}}}%
    }%
  }%
}
\newcommand{\cvheader}[2]{%
  \noindent
  \begin{minipage}[c]{\dimexpr\textwidth-2.7cm\relax}
    \raggedright
    {\fontsize{24pt}{28pt}\selectfont\bfseries #1\par}
    \vspace{4pt}
    {\small #2\par}
  \end{minipage}\hfill
  \begin{minipage}[c]{2.2cm}
    \centering\cvphotobox
  \end{minipage}%
  \par\vspace{2pt}
}
