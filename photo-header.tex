% Shared photo header for both resume languages.
\usepackage{graphicx}
% Upload photo.jpg (preferred) or photo.png to the project root.
% The image keeps its aspect ratio and fits within 2.7 cm x 3.5 cm.
\newcommand{\cvphotobox}{%
  \IfFileExists{photo.jpg}{%
    \includegraphics[width=2.7cm,height=3.5cm,keepaspectratio]{photo.jpg}%
  }{%
    \IfFileExists{photo.png}{%
      \includegraphics[width=2.7cm,height=3.5cm,keepaspectratio]{photo.png}%
    }{%
      {\setlength{\fboxsep}{0pt}\color{accent!40}%
      \fbox{\parbox[c][3.5cm][c]{2.7cm}{\centering
        \normalfont\small\color{accent} PHOTO\\[6pt]
        \scriptsize photo.jpg / photo.png}}}%
    }%
  }%
}
% Three columns: name | contact information | portrait.
\newcommand{\cvheader}[2]{%
  \noindent
  \begin{minipage}[t][3.55cm][t]{2.9cm}
    \vspace{2mm}
    \raggedright{\fontsize{25pt}{31pt}\selectfont\bfseries #1\par}
  \end{minipage}\hfill
  \begin{minipage}[t][3.55cm][t]{\dimexpr\textwidth-6.2cm\relax}
    \vspace{2mm}
    \raggedright{\small #2\par}
  \end{minipage}\hfill
  \begin{minipage}[t][3.55cm][t]{2.7cm}
    \vspace{0pt}\centering\cvphotobox
  \end{minipage}%
  \par\vspace{1pt}
  {\color{ruleblue}\hrule height 1.2pt}
  \vspace{1pt}
  {\color{rulegold}\hrule height 0.5pt}
  \par\vspace{2pt}
}
